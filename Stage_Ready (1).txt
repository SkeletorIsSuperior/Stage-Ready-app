<body>

  <h1>Stage Ready</h1>
  <p class="subtitle">Tap each item when it's set</p>

  <div class="checklist" id="checklist">
    <!-- Items get inserted here by JavaScript -->
  </div>

  <div id="status">NOT READY</div>
  <button id="reset">Reset All</button>

  <script>
    // 1. Define your departments (edit this list to match your theater)
    const departments = ["Lights", "Sound", "Props", "Set", "Actors"];

    // 2. Track the state of each one
    const state = {};
    departments.forEach(d => state[d] = false);

    // 3. Build the checklist UI from the list
    const checklist = document.getElementById("checklist");
    departments.forEach(name => {
      const row = document.createElement("div");
      row.className = "item";
      row.innerHTML = `
        <span>${name}</span>
        <button class="toggle" data-name="${name}">Set</button>
      `;
      checklist.appendChild(row);
    });

    // 4. Handle button clicks
    document.querySelectorAll(".toggle").forEach(btn => {
      btn.addEventListener("click", () => {
        const name = btn.dataset.name;
        state[name] = !state[name];               // flip the state
        btn.classList.toggle("ready", state[name]);
        btn.textContent = state[name] ? "✓ Ready" : "Set";
        updateStatus();
      });
    });

    // 5. Check if everything is ready
    function updateStatus() {
      const allReady = departments.every(d => state[d]);
      const statusEl = document.getElementById("status");
      if (allReady) {
        statusEl.textContent = "✅ STAGE READY";
        statusEl.classList.add("ready");
      } else {
        statusEl.textContent = "NOT READY";
        statusEl.classList.remove("ready");
      }
    }

    // 6. Reset button
    document.getElementById("reset").addEventListener("click", () => {
      departments.forEach(d => state[d] = false);
      document.querySelectorAll(".toggle").forEach(btn => {
        btn.classList.remove("ready");
        btn.textContent = "Set";
      });
      updateStatus();
    });
  </script>
